cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bo1_bundled_speex C)
endif()
get_filename_component(BO1_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/speex")
# Matching bundled configure template: short=16bits, int=32bits on ARM64/Windows.
set(SIZE16 short)
set(SIZE32 int)
configure_file("${BO1_ROOT}/src/speex/speex_config_types.h.in" "${CMAKE_CURRENT_BINARY_DIR}/include/speex/speex_config_types.h" @ONLY)
set(BO1_SPEEX_FILES nb_celp sb_celp lpc ltp lsp quant_lsp lsp_tables_nb gain_table gain_table_lbr cb_search filters bits modes speex vq high_lsp_tables vbr hexc_table exc_5_256_table exc_5_64_table exc_8_128_table exc_10_32_table exc_10_16_table exc_20_32_table hexc_10_32_table misc speex_header speex_callbacks math_approx stereo lbr_48k_tables)
set(BO1_SPEEX_SOURCES)
foreach(name IN LISTS BO1_SPEEX_FILES)
    list(APPEND BO1_SPEEX_SOURCES "${BO1_ROOT}/src/groupvoice/speex/${name}.c")
endforeach()
add_library(bo1_native_speex STATIC ${BO1_SPEEX_SOURCES})
target_include_directories(bo1_native_speex PUBLIC "${BO1_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}/include")
set_property(TARGET bo1_native_speex PROPERTY POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
    target_link_libraries(bo1_native_speex PUBLIC m)
endif()
