cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(bo1_native_vpx_dependency LANGUAGES C)
endif()
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Native VPX dependency requires Android ARM64")
endif()
get_filename_component(BO1_VPX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BO1_VPX_UPSTREAM "${BO1_VPX_ROOT}/third_party/libvpx-1.5.0")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(BO1_VPX_ADAPTED "${CMAKE_CURRENT_BINARY_DIR}/adapted")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/apply_native_adaptations.py" "${BO1_VPX_UPSTREAM}" "${BO1_VPX_ADAPTED}" RESULT_VARIABLE BO1_VPX_ADAPT_RESULT)
if(NOT BO1_VPX_ADAPT_RESULT EQUAL 0)
    message(FATAL_ERROR "Pinned native VPX source adaptation failed")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/sources.cmake")
set(BO1_VPX_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/generated/vpx_config.c")
foreach(BO1_VPX_SOURCE_NAME IN LISTS BO1_VPX_SOURCE_NAMES)
    if(EXISTS "${BO1_VPX_ADAPTED}/${BO1_VPX_SOURCE_NAME}")
        list(APPEND BO1_VPX_SOURCES "${BO1_VPX_ADAPTED}/${BO1_VPX_SOURCE_NAME}")
    else()
        list(APPEND BO1_VPX_SOURCES "${BO1_VPX_UPSTREAM}/${BO1_VPX_SOURCE_NAME}")
    endif()
endforeach()
add_library(bo1_native_vpx_codec STATIC ${BO1_VPX_SOURCES})
set_target_properties(bo1_native_vpx_codec PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
target_include_directories(bo1_native_vpx_codec PUBLIC "${BO1_VPX_UPSTREAM}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated" "${BO1_VPX_UPSTREAM}/vp8/encoder")
target_compile_options(bo1_native_vpx_codec PRIVATE -fwrapv -fno-strict-aliasing -fsigned-char -ffunction-sections -fdata-sections -fsanitize=undefined -fno-sanitize-recover=all)
# Real Bionic libc supplies pthread; Android has no split -lpthread archive.
target_link_libraries(bo1_native_vpx_codec PUBLIC m)
