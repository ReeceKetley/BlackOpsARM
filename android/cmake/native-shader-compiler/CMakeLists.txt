cmake_minimum_required(VERSION 3.22.1)
project(BlackOpsNativeShaderCompiler LANGUAGES C)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This integration requires Android ARM64")
endif()
get_filename_component(BO1_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(VKD3D_SOURCE "${BO1_ROOT}/third_party/native-shader-compiler/vkd3d-1.17")
if(NOT EXISTS "${VKD3D_SOURCE}/COPYING")
    message(FATAL_ERROR "Run tools/port/prepare_native_shader_compiler.py first")
endif()
find_program(BO1_BISON bison PATHS "C:/msys64/usr/bin" REQUIRED)
find_program(BO1_FLEX flex PATHS "C:/msys64/usr/bin" REQUIRED)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}")
foreach(parser hlsl preproc)
    add_custom_command(OUTPUT "${generated}/${parser}.tab.c" "${generated}/${parser}.tab.h"
        COMMAND "${BO1_BISON}" -d -o "${generated}/${parser}.tab.c" "${VKD3D_SOURCE}/libs/vkd3d-shader/${parser}.y"
        DEPENDS "${VKD3D_SOURCE}/libs/vkd3d-shader/${parser}.y" VERBATIM)
    add_custom_command(OUTPUT "${generated}/${parser}.yy.c"
        COMMAND "${BO1_FLEX}" -o "${generated}/${parser}.yy.c" "${VKD3D_SOURCE}/libs/vkd3d-shader/${parser}.l"
        DEPENDS "${VKD3D_SOURCE}/libs/vkd3d-shader/${parser}.l" "${generated}/${parser}.tab.h" VERBATIM)
    list(APPEND compiler_sources "${generated}/${parser}.tab.c" "${generated}/${parser}.yy.c")
endforeach()
include(CheckCSourceCompiles)
include(CheckIncludeFile)
include(CheckSymbolExists)
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
foreach(header pthread.h stdint.h inttypes.h unistd.h)
    string(TOUPPER "${header}" feature)
    string(REPLACE "." "_" feature "${feature}")
    check_include_file("${header}" HAVE_${feature})
endforeach()
check_symbol_exists(gettid "unistd.h" HAVE_GETTID)
check_c_source_compiles("int main(){return __builtin_popcount(3u);}" HAVE_BUILTIN_POPCOUNT)
check_c_source_compiles("int main(){return __builtin_clz(3u);}" HAVE_BUILTIN_CLZ)
check_c_source_compiles("int main(){unsigned c;return __builtin_add_overflow(1u,2u,&c);}" HAVE_BUILTIN_ADD_OVERFLOW)
check_c_source_compiles("int main(){int n=0;return __sync_add_and_fetch(&n,1);}" HAVE_SYNC_ADD_AND_FETCH)
check_c_source_compiles("int main(){int n=0;return __sync_bool_compare_and_swap(&n,0,1);}" HAVE_SYNC_BOOL_COMPARE_AND_SWAP)
check_c_source_compiles("int main(){int n=0;return __atomic_exchange_n(&n,1,__ATOMIC_SEQ_CST);}" HAVE_ATOMIC_EXCHANGE_N)
check_c_source_compiles("#include <pthread.h>\nint main(){return pthread_setname_np(pthread_self(),\"bo1-shader\");}" HAVE_PTHREAD_SETNAME_NP_2)
set(config "#pragma once\n#define _GNU_SOURCE 1\n#define PACKAGE_VERSION \"1.17\"\n#define HAVE_SPIRV_UNIFIED1_SPIRV_H 1\n#define HAVE_SPIRV_UNIFIED1_GLSL_STD_450_H 1\n")
foreach(feature HAVE_PTHREAD_H HAVE_STDINT_H HAVE_INTTYPES_H HAVE_UNISTD_H HAVE_GETTID
    HAVE_BUILTIN_POPCOUNT HAVE_BUILTIN_CLZ HAVE_BUILTIN_ADD_OVERFLOW
    HAVE_SYNC_ADD_AND_FETCH HAVE_SYNC_BOOL_COMPARE_AND_SWAP HAVE_ATOMIC_EXCHANGE_N HAVE_PTHREAD_SETNAME_NP_2)
    if(${feature})
        string(APPEND config "#define ${feature} 1\n")
    endif()
endforeach()
# Preserve timestamps when configuration is unchanged; regenerating the parent
# engine build must not recompile the entire shader compiler unnecessarily.
file(CONFIGURE OUTPUT "${generated}/config.h" CONTENT "${config}" @ONLY)
file(CONFIGURE OUTPUT "${generated}/vkd3d_version.h" CONTENT "#define VKD3D_VCS_ID \"\"\n" @ONLY)
foreach(unit checksum d3d_asm d3dbc dxbc dxil fx glsl hlsl hlsl_codegen hlsl_constant_ops ir msl spirv tpf vkd3d_shader_main)
    list(APPEND compiler_sources "${VKD3D_SOURCE}/libs/vkd3d-shader/${unit}.c")
endforeach()
foreach(unit debug error memory utf8)
    list(APPEND compiler_sources "${VKD3D_SOURCE}/libs/vkd3d-common/${unit}.c")
endforeach()
add_library(bo1_native_shader_compiler SHARED ${compiler_sources})
target_compile_features(bo1_native_shader_compiler PRIVATE c_std_11)
target_compile_definitions(bo1_native_shader_compiler PRIVATE LIBVKD3D_SHADER_SOURCE)
target_include_directories(bo1_native_shader_compiler PRIVATE "${generated}" "${VKD3D_SOURCE}/include/private"
    "${VKD3D_SOURCE}/libs/vkd3d-shader" "${BO1_ROOT}/third_party/native-shader-compiler/spirv-headers/include")
target_include_directories(bo1_native_shader_compiler PUBLIC "${VKD3D_SOURCE}/include")
target_link_libraries(bo1_native_shader_compiler PRIVATE m)
add_executable(bo1_native_shader_compiler_smoke "${BO1_ROOT}/tools/port/tests/native_shader_compiler_smoke.c")
target_link_libraries(bo1_native_shader_compiler_smoke PRIVATE bo1_native_shader_compiler)
